x1 = 0        % x1 = loop index i
x2 = 0        % x2 = accumulated output y
x3 = 0        % x3 = memory pointer
x4 = [x3]     % read N (size of input array) from address 0

.loop
x15 = x1 - x4 % compare index x1 with N
BEQ .exit     % exit loop if i == N

x3 = x3 + 4   % advance memory pointer to x_i
x5 = [x3]     % load input sample x_i from memory

% Determine weight based on index i and compute (w_i * x_i)
x15 = x1 - 0
BEQ .w0
x15 = x1 - 1
BEQ .w1
x15 = x1 - 2
BEQ .w2
x15 = x1 - 3
BEQ .w3

.w0
x6 = x5 * 1   % w0 = 1
BAL .add

.w1
x6 = x5 * 2   % w1 = 2
BAL .add

.w2
x6 = x5 * 3   % w2 = 3
BAL .add

.w3
x6 = x5 * 4   % w3 = 4

.add
x2 = x2 + x6  % accumulate into y
x1 = x1 + 1   % i++
BAL .loop

.exit
x3 = x3 + 4   % pointer moves to address 20 (0x14)
[x3] = x2     % store filtered output y at 0x14
